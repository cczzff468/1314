   143→/* 统计 + 筛选渲染列表 */
   144→async function loadBooks(body) {
   145→  const listEl = body.querySelector('#wb-list');
   146→  const statEl = body.querySelector('#wb-stat');
   147→  if (!listEl) return;
   148→  const books = await DB.all('worldbooks');
   149→  const entries = await DB.all('wbentries');
   150→  const byBook = new Map();
   151→  entries.forEach(e => {
   152→    if (e && e.bookId) {
   153→      if (!byBook.has(e.bookId)) byBook.set(e.bookId, []);
   154→      byBook.get(e.bookId).push(e);
   155→    }
   156→  });
   157→  books.sort((a, b) => (b.updatedAt || 0) - (a.updatedAt || 0));
   158→
   159→  applyStats(body, books, entries, statEl);
   160→
   161→  /* 筛选：范围 + 专属视角联系人 + 仅看已启用 */
   162→  let shown = books;
   163→  if (curScope !== 'all') shown = shown.filter(b => b.scope === curScope);
   164→  if (curScope === 'exclusive' && curContact) shown = shown.filter(b => Array.isArray(b.bound) && b.bound.includes(curContact));
   165→  if (onlyEnabled) shown = shown.filter(b => b.enabled);
   166→
   167→  listEl.innerHTML = '';
   168→  if (!books.length) {
   169→    listEl.innerHTML = `
   170→      <div class="empty-state">
   171→        ${BOOK_SVG.replace('width="19" height="19"', 'width="52" height="52"')}
   172→        <div class="es-title">没有世界书</div>
   173→        <div>点右上角 + 新建一个世界书</div>
   174→      </div>`;
   175→    return;
   176→  }
   177→  if (!shown.length) {
   178→    const ctTitle = curScope === 'exclusive' && curContact
   179→      ? `「${curContact}」暂无专属世界书`
   180→      : (onlyEnabled && curScope === 'all' ? '没有已启用的世界书' : '该范围下暂无世界书');
   181→    const ctSub = curScope === 'exclusive' && curContact
   182→      ? '在该角色的聊天设置里，或书籍的「绑定角色」里绑定后才会出现'
   183→      : (onlyEnabled && curScope === 'all' ? '打开卡片上的开关后，会出现在这里' : '切换范围筛选，或点 + 新建一个');
   184→    listEl.innerHTML = `
   185→      <div class="empty-state">
   186→        ${SCOPE_SVG.replace('width="15" height="15"', 'width="44" height="44"')}
   187→        <div class="es-title">${escapeHtml(ctTitle)}</div>
   188→        <div>${escapeHtml(ctSub)}</div>
   189→      </div>`;
   190→    return;
   191→  }
   192→  shown.forEach(b => listEl.appendChild(bookCard(b, byBook.get(b.id) || [], body)));
   193→}

   195→/* 更新统计面板数字 + 高亮 + 头部统计行（不重绘列表） */
   196→function applyStats(body, books, entries, statEl) {
   197→  const cnt = { global: 0, local: 0, exclusive: 0, enabled: 0 };
   198→  books.forEach(b => {
   199→    if (b.scope === 'global' || b.scope === 'local' || b.scope === 'exclusive') cnt[b.scope]++;
   200→    if (b.enabled) cnt.enabled++;
   201→  });
   202→  ['global', 'local', 'exclusive'].forEach(k => {
   203→    const cell = body.querySelector(`.wb-stat[data-st="${k}"]`);
   204→    if (cell) {
   205→      cell.querySelector('.wb-stat-num').textContent = cnt[k];
   206→      cell.classList.toggle('on', curScope === k);
   207→    }
   208→  });
   209→  const enCell = body.querySelector('.wb-stat[data-st="enabled"]');
   210→  if (enCell) {
   211→    enCell.querySelector('.wb-stat-num').textContent = cnt.enabled;
   212→    enCell.classList.toggle('on', onlyEnabled);
   213→  }
   214→  const pageEl = body.closest('.nav-page') || body;
   215→  pageEl.querySelectorAll('.wb-chip').forEach(c => c.classList.toggle('on', c.dataset.f === curScope));
   216→
   217→  /* 专属视角：右上角联系人下拉（仅专属筛选时显示；同步当前选择名） */
   218→  const ctBtn = pageEl.querySelector('.wb-ctbtn');
   219→  if (ctBtn) {
   220→    ctBtn.style.display = curScope === 'exclusive' ? '' : 'none';
   221→    const nm = ctBtn.querySelector('.wb-ct-name');
   222→    if (nm) nm.textContent = curContact || '全部角色';
   223→  }
   224→
   225→  if (statEl) {
   226→    if (books.length) {
   227→      let last = 0;
   228→      books.forEach(b => { last = Math.max(last, b.updatedAt || 0); });
   229→      entries.forEach(e => { last = Math.max(last, e.updatedAt || 0); });
   230→      statEl.textContent = `共 ${books.length} 个世界书 · 最后更新 ${fmtSmartTime(last)}`;
   231→    } else {
   232→      statEl.textContent = '还没有任何世界书';
   233→    }
   234→  }
   235→}

   246→function bookCard(book, entries, body) {
   247→  const kwCount = entries.reduce((n, e) => n + (Array.isArray(e.keywords) ? e.keywords.length : 0), 0);
   248→  const scope = SCOPES[book.scope] || SCOPES.global;
   249→  const card = el('div', 'wb-card' + (book.enabled ? '' : ' off'));
   250→  /* 启用开关在前，「编辑」「更多(⋯)」在开关右侧（同一行垂直对齐） */
   251→  card.innerHTML = `
   252→    <div class="wb-card-head">
   253→      <div class="wb-card-icon s-${book.scope}">${BOOK_SVG}</div>
   254→      <div class="wb-card-name ellipsis">${escapeHtml(book.name || '未命名')}</div>
   255→      <div class="wb-scope-chip s-${book.scope}">${scope.label}</div>
   256→      <div class="switch ${book.enabled ? 'on' : ''}" role="switch" aria-label="启用世界书"></div>
   257→      <button class="wb-act-btn edit">编辑</button>
   258→      <button class="wb-act-btn more" aria-label="更多操作">${MORE_SVG}</button>
   259→    </div>
   260→    <div class="wb-card-meta">
   261→      <div class="ellipsis">${entries.length} 个条目 · ${kwCount} 个关键词</div>
   262→      <div class="ellipsis">${scope.label}范围 · 更新于 ${fmtSmartTime(book.updatedAt)}</div>
   263→    </div>`;
   264→
   265→  card.querySelector('.switch').addEventListener('click', async (e) => {
   266→    e.stopPropagation();
   267→    haptic(4);
   268→    book.enabled = !book.enabled;
   269→    book.updatedAt = Date.now();
   270→    await DB.put('worldbooks', book);
   271→    e.currentTarget.classList.toggle('on', book.enabled);
   272→    card.classList.toggle('off', !book.enabled);
   273→    await refreshStatsOnly(body);
   274→    if (onlyEnabled && !book.enabled) await loadBooks(body);
   275→    toast(book.enabled ? '已启用' : '已停用');
   276→  });
   277→  card.querySelector('.wb-act-btn.edit').onclick = (e) => { e.stopPropagation(); haptic(4); openBookEntries(book); };
   278→  card.querySelector('.wb-act-btn.more').onclick = (e) => { e.stopPropagation(); haptic(4); bookMenu(book); };
   279→  /* 卡片其余区域点击 = 进入编辑 */
   280→  card.addEventListener('click', (e) => {
   281→    if (e.target.closest('.switch') || e.target.closest('.wb-act-btn')) return;
   282→    haptic(4);
   283→    openBookEntries(book);
   284→  });
   285→  return card;
   286→}

   288→async function bookMenu(book) {
   289→  const v = await actionSheet([
   290→    { text: book.enabled ? '停用世界书' : '启用世界书', value: 'toggle' },
   291→    { text: '重命名', value: 'rename' },
   292→    { text: '复制世界书', value: 'copy' },
   293→    { text: '删除世界书', value: 'del', danger: true },
   294→  ], { cls: 'wb-mono' });
   295→  if (!v) return;
   296→  if (v === 'toggle') {
   297→    book.enabled = !book.enabled;
   298→    book.updatedAt = Date.now();
   299→    await DB.put('worldbooks', book);
   300→    toast(book.enabled ? '已启用' : '已停用');
   301→  }
   302→  if (v === 'rename') {
   303→    const name = await promptDialog('重命名世界书', '', { value: book.name || '', okText: '好', cls: 'wb-mono' });
   304→    if (name && name.trim()) {
   305→      book.name = name.trim();
   306→      book.updatedAt = Date.now();
   307→      await DB.put('worldbooks', book);
   308→      toast('已重命名');
   309→    }
   310→  }
   311→  if (v === 'copy') {
   312→    const nid = uid('wb');
   313→    await DB.put('worldbooks', { ...book, id: nid, name: (book.name || '未命名') + ' 副本', enabled: false, createdAt: Date.now(), updatedAt: Date.now() });
   314→    const entries = await DB.byIndex('wbentries', 'bookId', book.id);
   315→    const copies = entries.map(e => ({ ...e, id: uid('wbe'), bookId: nid, updatedAt: Date.now() }));
   316→    if (copies.length) await DB.bulkPut('wbentries', copies);
   317→    toast('已复制（默认停用）');
   318→  }
   319→  if (v === 'del') {
   320→    const ok = await confirmDialog('删除世界书', `删除「${book.name || '未命名'}」及其全部条目？`, { okText: '删除', danger: true, cls: 'wb-mono' });
   321→    if (!ok) return;
   322→    const entries = await DB.byIndex('wbentries', 'bookId', book.id);
   323→    for (const e of entries) await DB.del('wbentries', e.id);
   324→    await DB.del('worldbooks', book.id);
   325→    toast('已删除');
   326→  }
   327→  const body = root && root.querySelector('.page-body');
   328→  if (body && body.querySelector('#wb-list')) await loadBooks(body);
   329→}

   674→/* 读取「信息」APP 好友列表（ios-im 库 kv 表 im.friends，localStorage 兜底；同源可直读） */
   675→function loadChatFriends() {
   676→  const fromLS = () => {
   677→    try {
   678→      const raw = localStorage.getItem('im.friends');
   679→      const arr = raw ? JSON.parse(raw) : [];
   680→      return (Array.isArray(arr) ? arr : []).filter(f => f && f.name).map(f => ({ id: f.id, name: f.name }));
   681→    } catch (e) { return []; }
   682→  };
   683→  return new Promise((resolve) => {
   684→    let done = false;
   685→    const finish = (v) => { if (!done) { done = true; resolve(v); } };
   686→    try {
   687→      const req = indexedDB.open('ios-im');
   688→      req.onupgradeneeded = () => {
   689→        try { if (!req.result.objectStoreNames.contains('kv')) req.result.createObjectStore('kv'); } catch (e) { /* noop */ }
   690→      };
   691→      req.onsuccess = () => {
   692→        const db = req.result;
   693→        try {
   694→          const r = db.transaction('kv', 'readonly').objectStore('kv').get('im.friends');
   695→          r.onsuccess = () => {
   696→            const arr = Array.isArray(r.result) ? r.result : null;
   697→            const list = (arr || []).filter(f => f && f.name).map(f => ({ id: f.id, name: f.name }));
   698→            finish(list.length ? list : fromLS());
   699→          };
   700→          r.onerror = () => finish(fromLS());
   701→        } catch (e) { finish(fromLS()); }
   702→      };
   703→      req.onerror = () => finish(fromLS());
   704→      setTimeout(() => finish(fromLS()), 1500); /* 兜底超时 */
   705→    } catch (e) { finish(fromLS()); }
   706→  });
   707→}
