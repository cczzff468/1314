    77→const getSynthVoices = (): Promise<SpeechSynthesisVoice[]> => {
    78→  const synth = window.speechSynthesis
    79→  const first = synth.getVoices()
    80→  if (first.length > 0) return Promise.resolve(first)
    81→  return new Promise((resolve) => {
    82→    const onChange = () => {
    83→      const list = synth.getVoices()
    84→      if (list.length > 0) {
    85→        synth.removeEventListener('voiceschanged', onChange)
    86→        resolve(list)
    87→      }
    88→    }
    89→    synth.addEventListener('voiceschanged', onChange)
    90→    window.setTimeout(() => {
    91→      synth.removeEventListener('voiceschanged', onChange)
    92→      resolve(synth.getVoices())
    93→    }, 800)
    94→  })
    95→}

    97→const pickSynthVoice = (voices: SpeechSynthesisVoice[], voiceId: string, lang: string) => {
    98→  const key = voiceId.toLowerCase().trim()
    99→  if (key) {
   100→    const exact = voices.find((v) => v.name.toLowerCase() === key)
   101→    if (exact) return exact
   102→    const byName = voices.find((v) => v.name.toLowerCase().includes(key))
   103→    if (byName) return byName
   104→  }
   105→  const want = (lang || 'zh').toLowerCase()
   106→  return voices.find((v) => v.lang.replace('_', '-').toLowerCase().startsWith(want))
   107→}

   325→  const runWebSttTest = async (lang: string): Promise<string> => {
   326→    const st = await micPermissionState()

   334→          w === 'denied'

   342→      }

   347→      showHint('正在聆听（浏览器 Web Speech API），请说一句话…')

   393→        wantWeb
   394→          ? `测试通过（已回退服务端识别）：${text.slice(0, 20)}${text.length > 20 ? '…' : ''}；建议引擎改用「服务端」或「自动」`
   395→          : `测试通过（服务端识别）：${text.slice(0, 24)}${text.length > 24 ? '…' : ''}`
   396→      )
   397→    } catch (e) {
   398→      const msg = String((e as Error)?.name || '') + ' ' + String((e as Error)?.message || '')

   400→        /NotAllowed|Permission|denied|拒绝/i.test(msg)

   407→      )

   409→      setSttTesting(false)
   410→    }
