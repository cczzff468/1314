    70→const getSynthVoices = (): Promise<SpeechSynthesisVoice[]> => {
    71→  const synth = window.speechSynthesis
    72→  const first = synth.getVoices()
    73→  if (first.length > 0) return Promise.resolve(first)
    74→  return new Promise((resolve) => {
    75→    const onChange = () => {
    76→      const list = synth.getVoices()
    77→      if (list.length > 0) {
    78→        synth.removeEventListener('voiceschanged', onChange)
    79→        resolve(list)
    80→      }
    81→    }
    82→    synth.addEventListener('voiceschanged', onChange)
    83→    window.setTimeout(() => {
    84→      synth.removeEventListener('voiceschanged', onChange)
    85→      resolve(synth.getVoices())
    86→    }, 800)
    87→  })
    88→}

    90→const pickSynthVoice = (voices: SpeechSynthesisVoice[], voiceId: string, lang: string) => {
    91→  const key = voiceId.toLowerCase().trim()
    92→  if (key) {
    93→    const exact = voices.find((v) => v.name.toLowerCase() === key)
    94→    if (exact) return exact
    95→    const byName = voices.find((v) => v.name.toLowerCase().includes(key))
    96→    if (byName) return byName
    97→  }
    98→  const want = (lang || 'zh').toLowerCase()
    99→  return voices.find((v) => v.lang.replace('_', '-').toLowerCase().startsWith(want))
   100→}
