   130→/* 统计 + 筛选渲染列表 */
   131→async function loadBooks(body) {
   132→  const listEl = body.querySelector('#wb-list');
   133→  const statEl = body.querySelector('#wb-stat');
   134→  if (!listEl) return;
   135→  const books = await DB.all('worldbooks');
   136→  const entries = await DB.all('wbentries');
   137→  const byBook = new Map();
   138→  entries.forEach(e => {
   139→    if (e && e.bookId) {
   140→      if (!byBook.has(e.bookId)) byBook.set(e.bookId, []);
   141→      byBook.get(e.bookId).push(e);
   142→    }
   143→  });
   144→  books.sort((a, b) => (b.updatedAt || 0) - (a.updatedAt || 0));
   145→
   146→  applyStats(body, books, entries, statEl);
   147→
   148→  /* 筛选：范围 + 仅看已启用 */
   149→  let shown = books;
   150→  if (curScope !== 'all') shown = shown.filter(b => b.scope === curScope);
   151→  if (onlyEnabled) shown = shown.filter(b => b.enabled);
   152→
   153→  listEl.innerHTML = '';
   154→  if (!books.length) {
   155→    listEl.innerHTML = `
   156→      <div class="empty-state">
   157→        ${BOOK_SVG.replace('width="19" height="19"', 'width="52" height="52"')}
   158→        <div class="es-title">没有世界书</div>
   159→        <div>点右上角 + 新建一个世界书</div>
   160→      </div>`;
   161→    return;
   162→  }
   163→  if (!shown.length) {
   164→    listEl.innerHTML = `
   165→      <div class="empty-state">
   166→        ${SCOPE_SVG.replace('width="15" height="15"', 'width="44" height="44"')}
   167→        <div class="es-title">${onlyEnabled && curScope === 'all' ? '没有已启用的世界书' : '该范围下暂无世界书'}</div>
   168→        <div>${onlyEnabled && curScope === 'all' ? '打开卡片上的开关后，会出现在这里' : '切换范围筛选，或点 + 新建一个'}</div>
   169→      </div>`;
   170→    return;
   171→  }
   172→  shown.forEach(b => listEl.appendChild(bookCard(b, byBook.get(b.id) || [], body)));
   173→}

   175→/* 更新统计面板数字 + 高亮 + 头部统计行（不重绘列表） */
   176→function applyStats(body, books, entries, statEl) {
   177→  const cnt = { global: 0, local: 0, exclusive: 0, enabled: 0 };
   178→  books.forEach(b => {
   179→    if (b.scope === 'global' || b.scope === 'local' || b.scope === 'exclusive') cnt[b.scope]++;
   180→    if (b.enabled) cnt.enabled++;
   181→  });
   182→  ['global', 'local', 'exclusive'].forEach(k => {
   183→    const cell = body.querySelector(`.wb-stat[data-st="${k}"]`);
   184→    if (cell) {
   185→      cell.querySelector('.wb-stat-num').textContent = cnt[k];
   186→      cell.classList.toggle('on', curScope === k);
   187→    }
   188→  });
   189→  const enCell = body.querySelector('.wb-stat[data-st="enabled"]');
   190→  if (enCell) {
   191→    enCell.querySelector('.wb-stat-num').textContent = cnt.enabled;
   192→    enCell.classList.toggle('on', onlyEnabled);
   193→  }
   194→  body.querySelectorAll('.wb-chip').forEach(c => c.classList.toggle('on', c.dataset.f === curScope));
   195→
   196→  if (statEl) {
   197→    if (books.length) {
   198→      let last = 0;
   199→      books.forEach(b => { last = Math.max(last, b.updatedAt || 0); });
   200→      entries.forEach(e => { last = Math.max(last, e.updatedAt || 0); });
   201→      statEl.textContent = `共 ${books.length} 个世界书 · 最后更新 ${fmtSmartTime(last)}`;
   202→    } else {
   203→      statEl.textContent = '还没有任何世界书';
   204→    }
   205→  }
   206→}

   209→async function refreshStatsOnly(body) {
   210→  const statEl = body.querySelector('#wb-stat');
   211→  if (!statEl) return;
   212→  const books = await DB.all('worldbooks');
   213→  const entries = await DB.all('wbentries');
   214→  applyStats(body, books, entries, statEl);
   215→}

   217→function bookCard(book, entries, body) {
   218→  const kwCount = entries.reduce((n, e) => n + (Array.isArray(e.keywords) ? e.keywords.length : 0), 0);
   219→  const scope = SCOPES[book.scope] || SCOPES.global;
   220→  const card = el('div', 'wb-card' + (book.enabled ? '' : ' off'));
   221→  card.innerHTML = `
   222→    <div class="wb-card-head">
   223→      <div class="wb-card-icon s-${book.scope}">${BOOK_SVG}</div>
   224→      <div class="wb-card-name ellipsis">${escapeHtml(book.name || '未命名')}</div>
   225→      <div class="wb-scope-chip s-${book.scope}">${scope.label}</div>
   226→      <div class="switch ${book.enabled ? 'on' : ''}" role="switch" aria-label="启用世界书"></div>
   227→    </div>
   228→    <div class="wb-card-meta">
   229→      <div class="ellipsis">${entries.length} 个条目 · ${kwCount} 个关键词</div>
   230→      <div class="ellipsis">${scope.label}范围 · 更新于 ${fmtSmartTime(book.updatedAt)}</div>
   231→    </div>
   232→    <div class="wb-card-actions">
   233→      <button class="wb-act-btn edit">编辑</button>
   234→      <button class="wb-act-btn more" aria-label="更多操作">${MORE_SVG}</button>
   235→    </div>`;
   236→
   237→  card.querySelector('.switch').addEventListener('click', async (e) => {
   238→    e.stopPropagation();
   239→    haptic(4);
   240→    book.enabled = !book.enabled;
   241→    book.updatedAt = Date.now();
   242→    await DB.put('worldbooks', book);
   243→    e.currentTarget.classList.toggle('on', book.enabled);
   244→    card.classList.toggle('off', !book.enabled);
   245→    await refreshStatsOnly(body);
   246→    if (onlyEnabled && !book.enabled) await loadBooks(body);
   247→    toast(book.enabled ? '已启用' : '已停用');
   248→  });
   249→  card.querySelector('.wb-act-btn.edit').onclick = (e) => { e.stopPropagation(); haptic(4); openBookEntries(book); };
   250→  card.querySelector('.wb-act-btn.more').onclick = (e) => { e.stopPropagation(); haptic(4); bookMenu(book); };
   251→  /* 卡片其余区域点击 = 进入编辑 */
   252→  card.addEventListener('click', (e) => {
   253→    if (e.target.closest('.switch') || e.target.closest('.wb-act-btn')) return;
   254→    haptic(4);
   255→    openBookEntries(book);
   256→  });
   257→  return card;
   258→}

   260→async function bookMenu(book) {
   261→  const v = await actionSheet([
   262→    { text: book.enabled ? '停用世界书' : '启用世界书', value: 'toggle' },
   263→    { text: '重命名', value: 'rename' },
   264→    { text: '复制世界书', value: 'copy' },
   265→    { text: '删除世界书', value: 'del', danger: true },
   266→  ]);
   267→  if (!v) return;
   268→  if (v === 'toggle') {
   269→    book.enabled = !book.enabled;
   270→    book.updatedAt = Date.now();
   271→    await DB.put('worldbooks', book);
   272→    toast(book.enabled ? '已启用' : '已停用');
   273→  }
   274→  if (v === 'rename') {
   275→    const name = await promptDialog('重命名世界书', '', { value: book.name || '', okText: '好' });
   276→    if (name && name.trim()) {
   277→      book.name = name.trim();
   278→      book.updatedAt = Date.now();
   279→      await DB.put('worldbooks', book);
   280→      toast('已重命名');
   281→    }
   282→  }
   283→  if (v === 'copy') {
   284→    const nid = uid('wb');
   285→    await DB.put('worldbooks', { ...book, id: nid, name: (book.name || '未命名') + ' 副本', enabled: false, createdAt: Date.now(), updatedAt: Date.now() });
   286→    const entries = await DB.byIndex('wbentries', 'bookId', book.id);
   287→    const copies = entries.map(e => ({ ...e, id: uid('wbe'), bookId: nid, updatedAt: Date.now() }));
   288→    if (copies.length) await DB.bulkPut('wbentries', copies);
   289→    toast('已复制（默认停用）');
   290→  }
   291→  if (v === 'del') {
   292→    const ok = await confirmDialog('删除世界书', `删除「${book.name || '未命名'}」及其全部条目？`, { okText: '删除', danger: true });
   293→    if (!ok) return;
   294→    const entries = await DB.byIndex('wbentries', 'bookId', book.id);
   295→    for (const e of entries) await DB.del('wbentries', e.id);
   296→    await DB.del('worldbooks', book.id);
   297→    toast('已删除');
   298→  }
   299→  const body = root && root.querySelector('.page-body');
   300→  if (body && body.querySelector('#wb-list')) await loadBooks(body);
   301→}

   627→/* 读取「信息」APP 好友列表（ios-im 库 kv 表 im.friends，localStorage 兜底；同源可直读） */
   628→function loadChatFriends() {
   629→  const fromLS = () => {
   630→    try {
   631→      const raw = localStorage.getItem('im.friends');
   632→      const arr = raw ? JSON.parse(raw) : [];
   633→      return (Array.isArray(arr) ? arr : []).filter(f => f && f.name).map(f => ({ id: f.id, name: f.name }));
   634→    } catch (e) { return []; }
   635→  };
   636→  return new Promise((resolve) => {
   637→    let done = false;
   638→    const finish = (v) => { if (!done) { done = true; resolve(v); } };
   639→    try {
   640→      const req = indexedDB.open('ios-im');
   641→      req.onupgradeneeded = () => {
   642→        try { if (!req.result.objectStoreNames.contains('kv')) req.result.createObjectStore('kv'); } catch (e) { /* noop */ }
   643→      };
   644→      req.onsuccess = () => {
   645→        const db = req.result;
   646→        try {
   647→          const r = db.transaction('kv', 'readonly').objectStore('kv').get('im.friends');
   648→          r.onsuccess = () => {
   649→            const arr = Array.isArray(r.result) ? r.result : null;
   650→            const list = (arr || []).filter(f => f && f.name).map(f => ({ id: f.id, name: f.name }));
   651→            finish(list.length ? list : fromLS());
   652→          };
   653→          r.onerror = () => finish(fromLS());
   654→        } catch (e) { finish(fromLS()); }
   655→      };
   656→      req.onerror = () => finish(fromLS());
   657→      setTimeout(() => finish(fromLS()), 1500); /* 兜底超时 */
   658→    } catch (e) { finish(fromLS()); }
   659→  });
   660→}
