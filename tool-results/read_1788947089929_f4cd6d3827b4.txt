   135→/* 统计 + 筛选渲染列表 */
   136→async function loadBooks(body) {
   137→  const listEl = body.querySelector('#wb-list');
   138→  const statEl = body.querySelector('#wb-stat');
   139→  if (!listEl) return;
   140→  const books = await DB.all('worldbooks');
   141→  const entries = await DB.all('wbentries');
   142→  const byBook = new Map();
   143→  entries.forEach(e => {
   144→    if (e && e.bookId) {
   145→      if (!byBook.has(e.bookId)) byBook.set(e.bookId, []);
   146→      byBook.get(e.bookId).push(e);
   147→    }
   148→  });
   149→  books.sort((a, b) => (b.updatedAt || 0) - (a.updatedAt || 0));
   150→
   151→  applyStats(body, books, entries, statEl);
   152→
   153→  /* 筛选：范围 + 仅看已启用 */
   154→  let shown = books;
   155→  if (curScope !== 'all') shown = shown.filter(b => b.scope === curScope);
   156→  if (onlyEnabled) shown = shown.filter(b => b.enabled);
   157→
   158→  listEl.innerHTML = '';
   159→  if (!books.length) {
   160→    listEl.innerHTML = `
   161→      <div class="empty-state">
   162→        ${BOOK_SVG.replace('width="19" height="19"', 'width="52" height="52"')}
   163→        <div class="es-title">没有世界书</div>
   164→        <div>点右上角 + 新建一个世界书</div>
   165→      </div>`;
   166→    return;
   167→  }
   168→  if (!shown.length) {
   169→    listEl.innerHTML = `
   170→      <div class="empty-state">
   171→        ${SCOPE_SVG.replace('width="15" height="15"', 'width="44" height="44"')}
   172→        <div class="es-title">${onlyEnabled && curScope === 'all' ? '没有已启用的世界书' : '该范围下暂无世界书'}</div>
   173→        <div>${onlyEnabled && curScope === 'all' ? '打开卡片上的开关后，会出现在这里' : '切换范围筛选，或点 + 新建一个'}</div>
   174→      </div>`;
   175→    return;
   176→  }
   177→  shown.forEach(b => listEl.appendChild(bookCard(b, byBook.get(b.id) || [], body)));
   178→}

   180→/* 更新统计面板数字 + 高亮 + 头部统计行（不重绘列表） */
   181→function applyStats(body, books, entries, statEl) {
   182→  const cnt = { global: 0, local: 0, exclusive: 0, enabled: 0 };
   183→  books.forEach(b => {
   184→    if (b.scope === 'global' || b.scope === 'local' || b.scope === 'exclusive') cnt[b.scope]++;
   185→    if (b.enabled) cnt.enabled++;
   186→  });
   187→  ['global', 'local', 'exclusive'].forEach(k => {
   188→    const cell = body.querySelector(`.wb-stat[data-st="${k}"]`);
   189→    if (cell) {
   190→      cell.querySelector('.wb-stat-num').textContent = cnt[k];
   191→      cell.classList.toggle('on', curScope === k);
   192→    }
   193→  });
   194→  const enCell = body.querySelector('.wb-stat[data-st="enabled"]');
   195→  if (enCell) {
   196→    enCell.querySelector('.wb-stat-num').textContent = cnt.enabled;
   197→    enCell.classList.toggle('on', onlyEnabled);
   198→  }
   199→  const pageEl = body.closest('.nav-page') || body;
   200→  pageEl.querySelectorAll('.wb-chip').forEach(c => c.classList.toggle('on', c.dataset.f === curScope));
   201→
   202→  if (statEl) {
   203→    if (books.length) {
   204→      let last = 0;
   205→      books.forEach(b => { last = Math.max(last, b.updatedAt || 0); });
   206→      entries.forEach(e => { last = Math.max(last, e.updatedAt || 0); });
   207→      statEl.textContent = `共 ${books.length} 个世界书 · 最后更新 ${fmtSmartTime(last)}`;
   208→    } else {
   209→      statEl.textContent = '还没有任何世界书';
   210→    }
   211→  }
   212→}

   215→async function refreshStatsOnly(body) {

   217→  if (!statEl) return;
   218→  const books = await DB.all('worldbooks');
   219→  const entries = await DB.all('wbentries');
   220→  applyStats(body, books, entries, statEl);
   221→}

   223→function bookCard(book, entries, body) {
   224→  const kwCount = entries.reduce((n, e) => n + (Array.isArray(e.keywords) ? e.keywords.length : 0), 0);
   225→  const scope = SCOPES[book.scope] || SCOPES.global;
   226→  const card = el('div', 'wb-card' + (book.enabled ? '' : ' off'));
   227→  /* 编辑与「更多」上移至头部行，与启用开关垂直对齐 */
   228→  card.innerHTML = `
   229→    <div class="wb-card-head">
   230→      <div class="wb-card-icon s-${book.scope}">${BOOK_SVG}</div>
   231→      <div class="wb-card-name ellipsis">${escapeHtml(book.name || '未命名')}</div>
   232→      <div class="wb-scope-chip s-${book.scope}">${scope.label}</div>
   233→      <button class="wb-act-btn edit">编辑</button>
   234→      <button class="wb-act-btn more" aria-label="更多操作">${MORE_SVG}</button>
   235→      <div class="switch ${book.enabled ? 'on' : ''}" role="switch" aria-label="启用世界书"></div>
   236→    </div>
   237→    <div class="wb-card-meta">
   238→      <div class="ellipsis">${entries.length} 个条目 · ${kwCount} 个关键词</div>
   239→      <div class="ellipsis">${scope.label}范围 · 更新于 ${fmtSmartTime(book.updatedAt)}</div>
   240→    </div>`;
   241→
   242→  card.querySelector('.switch').addEventListener('click', async (e) => {
   243→    e.stopPropagation();
   244→    haptic(4);
   245→    book.enabled = !book.enabled;
   246→    book.updatedAt = Date.now();
   247→    await DB.put('worldbooks', book);
   248→    e.currentTarget.classList.toggle('on', book.enabled);
   249→    card.classList.toggle('off', !book.enabled);
   250→    await refreshStatsOnly(body);
   251→    if (onlyEnabled && !book.enabled) await loadBooks(body);
   252→    toast(book.enabled ? '已启用' : '已停用');
   253→  });
   254→  card.querySelector('.wb-act-btn.edit').onclick = (e) => { e.stopPropagation(); haptic(4); openBookEntries(book); };
   255→  card.querySelector('.wb-act-btn.more').onclick = (e) => { e.stopPropagation(); haptic(4); bookMenu(book); };
   256→  /* 卡片其余区域点击 = 进入编辑 */
   257→  card.addEventListener('click', (e) => {
   258→    if (e.target.closest('.switch') || e.target.closest('.wb-act-btn')) return;
   259→    haptic(4);
   260→    openBookEntries(book);
   261→  });
   262→  return card;
   263→}

   265→async function bookMenu(book) {
   266→  const v = await actionSheet([
   267→    { text: book.enabled ? '停用世界书' : '启用世界书', value: 'toggle' },
   268→    { text: '重命名', value: 'rename' },
   269→    { text: '复制世界书', value: 'copy' },
   270→    { text: '删除世界书', value: 'del', danger: true },
   271→  ]);
   272→  if (!v) return;
   273→  if (v === 'toggle') {
   274→    book.enabled = !book.enabled;
   275→    book.updatedAt = Date.now();
   276→    await DB.put('worldbooks', book);
   277→    toast(book.enabled ? '已启用' : '已停用');
   278→  }
   279→  if (v === 'rename') {
   280→    const name = await promptDialog('重命名世界书', '', { value: book.name || '', okText: '好' });
   281→    if (name && name.trim()) {
   282→      book.name = name.trim();
   283→      book.updatedAt = Date.now();
   284→      await DB.put('worldbooks', book);
   285→      toast('已重命名');
   286→    }
   287→  }
   288→  if (v === 'copy') {
   289→    const nid = uid('wb');
   290→    await DB.put('worldbooks', { ...book, id: nid, name: (book.name || '未命名') + ' 副本', enabled: false, createdAt: Date.now(), updatedAt: Date.now() });
   291→    const entries = await DB.byIndex('wbentries', 'bookId', book.id);
   292→    const copies = entries.map(e => ({ ...e, id: uid('wbe'), bookId: nid, updatedAt: Date.now() }));
   293→    if (copies.length) await DB.bulkPut('wbentries', copies);
   294→    toast('已复制（默认停用）');
   295→  }
   296→  if (v === 'del') {
   297→    const ok = await confirmDialog('删除世界书', `删除「${book.name || '未命名'}」及其全部条目？`, { okText: '删除', danger: true });
   298→    if (!ok) return;
   299→    const entries = await DB.byIndex('wbentries', 'bookId', book.id);
   300→    for (const e of entries) await DB.del('wbentries', e.id);
   301→    await DB.del('worldbooks', book.id);
   302→    toast('已删除');
   303→  }
   304→  const body = root && root.querySelector('.page-body');
   305→  if (body && body.querySelector('#wb-list')) await loadBooks(body);
   306→}

   633→/* 读取「信息」APP 好友列表（ios-im 库 kv 表 im.friends，localStorage 兜底；同源可直读） */
   634→function loadChatFriends() {
   635→  const fromLS = () => {
   636→    try {
   637→      const raw = localStorage.getItem('im.friends');
   638→      const arr = raw ? JSON.parse(raw) : [];
   639→      return (Array.isArray(arr) ? arr : []).filter(f => f && f.name).map(f => ({ id: f.id, name: f.name }));
   640→    } catch (e) { return []; }
   641→  };
   642→  return new Promise((resolve) => {
   643→    let done = false;
   644→    const finish = (v) => { if (!done) { done = true; resolve(v); } };
   645→    try {
   646→      const req = indexedDB.open('ios-im');
   647→      req.onupgradeneeded = () => {
   648→        try { if (!req.result.objectStoreNames.contains('kv')) req.result.createObjectStore('kv'); } catch (e) { /* noop */ }
   649→      };
   650→      req.onsuccess = () => {
   651→        const db = req.result;
   652→        try {
   653→          const r = db.transaction('kv', 'readonly').objectStore('kv').get('im.friends');
   654→          r.onsuccess = () => {
   655→            const arr = Array.isArray(r.result) ? r.result : null;
   656→            const list = (arr || []).filter(f => f && f.name).map(f => ({ id: f.id, name: f.name }));
   657→            finish(list.length ? list : fromLS());
   658→          };
   659→          r.onerror = () => finish(fromLS());
   660→        } catch (e) { finish(fromLS()); }
   661→      };
   662→      req.onerror = () => finish(fromLS());
   663→      setTimeout(() => finish(fromLS()), 1500); /* 兜底超时 */
   664→    } catch (e) { finish(fromLS()); }
   665→  });
   666→}
