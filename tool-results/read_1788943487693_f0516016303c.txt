    78→async function loadBooks(body) {
    79→  const listEl = body.querySelector('#wb-list');
    80→  const statEl = body.querySelector('#wb-stat');
    81→  if (!listEl) return;
    82→  const books = await DB.all('worldbooks');
    83→  const entries = await DB.all('wbentries');
    84→  const byBook = new Map();
    85→  entries.forEach(e => {
    86→    if (e && e.bookId) {
    87→      if (!byBook.has(e.bookId)) byBook.set(e.bookId, []);
    88→      byBook.get(e.bookId).push(e);
    89→    }
    90→  });
    91→  books.sort((a, b) => (b.updatedAt || 0) - (a.updatedAt || 0));
    92→
    93→  /* 统计行：共 N 个世界书 · 最后更新: X */
    94→  if (books.length) {
    95→    let last = 0;
    96→    books.forEach(b => { last = Math.max(last, b.updatedAt || 0); });
    97→    entries.forEach(e => { last = Math.max(last, e.updatedAt || 0); });
    98→    statEl.textContent = `共 ${books.length} 个世界书 · 最后更新: ${fmtSmartTime(last)}`;
    99→  } else {
   100→    statEl.textContent = '还没有任何世界书';
   101→  }
   102→
   103→  listEl.innerHTML = '';
   104→  if (!books.length) {
   105→    listEl.innerHTML = `
   106→      <div class="empty-state">
   107→        ${BOOK_SVG.replace('width="19" height="19"', 'width="52" height="52"')}
   108→        <div class="es-title">没有世界书</div>
   109→        <div>点右上角 + 新建一个世界书</div>
   110→      </div>`;
   111→    return;
   112→  }
   113→  books.forEach(b => listEl.appendChild(bookCard(b, byBook.get(b.id) || [])));
   114→}

   116→function bookCard(book, entries) {
   117→  const kwCount = entries.reduce((n, e) => n + (Array.isArray(e.keywords) ? e.keywords.length : 0), 0);
   118→  const scope = SCOPES[book.scope] || SCOPES.global;
   119→  const card = el('div', 'wb-card' + (book.enabled ? '' : ' off'));
   120→  card.innerHTML = `
   121→    <div class="wb-card-head">
   122→      <div class="wb-card-icon">${BOOK_SVG}</div>
   123→      <div class="wb-card-name ellipsis">${escapeHtml(book.name || '未命名')}</div>
   124→      <div class="wb-scope-chip s-${book.scope}">${scope.label}</div>
   125→      <div class="switch ${book.enabled ? 'on' : ''}" role="switch" aria-label="启用世界书"></div>
   126→    </div>
   127→    <div class="wb-card-meta">
   128→      <div>条目: ${entries.length} · 关键词: ${kwCount}</div>
   129→      <div>范围: ${scope.label} · 更新: ${fmtSmartTime(book.updatedAt)}</div>
   130→    </div>
   131→    <div class="wb-card-actions">
   132→      <button class="wb-act-btn edit">编辑</button>
   133→      <button class="wb-act-btn more" aria-label="更多操作">${MORE_SVG}</button>
   134→    </div>`;
   135→
   136→  card.querySelector('.switch').addEventListener('click', async (e) => {
   137→    e.stopPropagation();
   138→    haptic(4);
   139→    book.enabled = !book.enabled;
   140→    book.updatedAt = Date.now();
   141→    await DB.put('worldbooks', book);
   142→    e.currentTarget.classList.toggle('on', book.enabled);
   143→    card.classList.toggle('off', !book.enabled);
   144→    statRefresh(book);
   145→    toast(book.enabled ? '已启用' : '已停用');
   146→  });
   147→  card.querySelector('.wb-act-btn.edit').onclick = (e) => { e.stopPropagation(); haptic(4); openBookEntries(book); };
   148→  card.querySelector('.wb-act-btn.more').onclick = (e) => { e.stopPropagation(); haptic(4); bookMenu(book); };
   149→  /* 卡片其余区域点击 = 进入编辑 */
   150→  card.addEventListener('click', (e) => {
   151→    if (e.target.closest('.switch') || e.target.closest('.wb-act-btn')) return;
   152→    haptic(4);
   153→    openBookEntries(book);
   154→  });
   155→  return card;
   156→}

   176→async function bookMenu(book) {
   177→  const v = await actionSheet([
   178→    { text: book.enabled ? '停用世界书' : '启用世界书', value: 'toggle' },
   179→    { text: '重命名', value: 'rename' },
   180→    { text: '复制世界书', value: 'copy' },
   181→    { text: '删除世界书', value: 'del', danger: true },
   182→  ]);
   183→  if (!v) return;
   184→  if (v === 'toggle') {
   185→    book.enabled = !book.enabled;
   186→    book.updatedAt = Date.now();
   187→    await DB.put('worldbooks', book);
   188→    toast(book.enabled ? '已启用' : '已停用');
   189→  }
   190→  if (v === 'rename') {
   191→    const name = await promptDialog('重命名世界书', '', { value: book.name || '', okText: '好' });
   192→    if (name && name.trim()) {
   193→      book.name = name.trim();
   194→      book.updatedAt = Date.now();
   195→      await DB.put('worldbooks', book);
   196→      toast('已重命名');
   197→    }
   198→  }
   199→  if (v === 'copy') {
   200→    const nid = uid('wb');
   201→    await DB.put('worldbooks', { ...book, id: nid, name: (book.name || '未命名') + ' 副本', enabled: false, createdAt: Date.now(), updatedAt: Date.now() });
   202→    const entries = await DB.byIndex('wbentries', 'bookId', book.id);
   203→    const copies = entries.map(e => ({ ...e, id: uid('wbe'), bookId: nid, updatedAt: Date.now() }));
   204→    if (copies.length) await DB.bulkPut('wbentries', copies);
   205→    toast('已复制（默认停用）');
   206→  }
   207→  if (v === 'del') {
   208→    const ok = await confirmDialog('删除世界书', `删除「${book.name || '未命名'}」及其全部条目？`, { okText: '删除', danger: true });
   209→    if (!ok) return;
   210→    const entries = await DB.byIndex('wbentries', 'bookId', book.id);
   211→    for (const e of entries) await DB.del('wbentries', e.id);
   212→    await DB.del('worldbooks', book.id);
   213→    toast('已删除');
   214→  }
   215→  const body = root && root.querySelector('.page-body');
   216→  if (body && body.querySelector('#wb-list')) await loadBooks(body);
   217→}

   374→async function loadEntries(book, body) {
   375→  const listEl = body.querySelector('#wbe-list');
   376→  if (!listEl) return;
   377→  const boundVal = body.querySelector('#wbe-bound-val');
   378→  if (boundVal) {
   379→    const names = Array.isArray(book.bound) ? book.bound.filter(Boolean) : [];
   380→    boundVal.textContent = names.length ? names.join('、') : '未绑定';
   381→    boundVal.classList.toggle('tappable', !names.length);
   382→  }
   383→  const entries = await DB.byIndex('wbentries', 'bookId', book.id);
   384→  entries.sort((a, b) => (b.priority || 0) - (a.priority || 0));
   385→
   386→  listEl.innerHTML = '';
   387→  if (!entries.length) {
   388→    listEl.innerHTML = `
   389→      <div class="empty-state">

   426→async function entryMenu(book, entry) {
   427→  const v = await actionSheet([
   428→    { text: '编辑条目', value: 'edit' },
   429→    { text: entry.enabled ? '停用条目' : '启用条目', value: 'toggle' },

   434→  if (v === 'toggle') {
   435→    entry.enabled = !entry.enabled;
   436→    entry.updatedAt = Date.now();
   437→    book.updatedAt = Date.now();
   438→    await DB.put('wbentries', entry);
   439→    await DB.put('worldbooks', book);
   440→    toast(entry.enabled ? '条目已启用' : '条目已停用');
   441→  }
   442→  if (v === 'del') {
   443→    const ok = await confirmDialog('删除条目', `删除「${entry.name || '未命名条目'}」？`, { okText: '删除', danger: true });
   444→    if (!ok) return;
   445→    await DB.del('wbentries', entry.id);
   446→    book.updatedAt = Date.now();
   447→    await DB.put('worldbooks', book);
   448→    toast('已删除');
   449→  }
   450→  await refreshEntriesView(book);
   451→}

   453→/* ============ 绑定角色（专属范围：多选信息APP好友） ============ */
   454→function openBindPicker(book, pageEl) {
   455→  sheet({
   456→    title: '绑定角色',
   457→    build(body, close) {
   458→      body.classList.add('wb-bind-body');
   459→      const tip = el('div', 'wb-bind-tip');
   460→      tip.textContent = '专属世界书仅在与绑定角色的对话中生效';
   461→      body.appendChild(tip);
   462→      loadChatFriends().then(friends => {
   463→        if (!friends.length) {
   464→          const empty = el('div', 'wb-bind-empty');
   465→          empty.textContent = '还没有角色，去「信息」APP 添加好友后再来绑定';
   466→          body.appendChild(empty);
   467→          return;
   468→        }
   469→        friends.forEach(f => {
   470→          const on = (Array.isArray(book.bound) ? book.bound : []).includes(f.name);

   472→          row.innerHTML = `<div class="wb-bind-check"></div><div class="wb-bind-name ellipsis">${escapeHtml(f.name)}</div>`;
   473→          row.onclick = () => {
   474→            haptic(4);
   475→            const list = Array.isArray(book.bound) ? book.bound : [];
   476→            const i = list.indexOf(f.name);
   477→            if (i >= 0) list.splice(i, 1); else list.push(f.name);
   478→            book.bound = list;
   479→            row.classList.toggle('on', list.includes(f.name));
   480→          };
   481→          body.appendChild(row);
   482→        });
   483→      });
   484→      const actions = el('div', 'sheet-actions');
   485→      actions.style.paddingTop = '12px';
   486→      const done = el('button', 'btn-fill');
   487→      done.type = 'button';
   488→      done.textContent = '完成';
   489→      done.onclick = () => close();
   490→      actions.appendChild(done);
   491→      body.appendChild(actions);
   492→    },
   493→    onClose: async () => {
   494→      book.updatedAt = Date.now();
   495→      await DB.put('worldbooks', book);
   496→      await refreshEntriesView(book);
   497→      toast('绑定已保存');
   498→    },
   499→  });
   500→}

   502→/* 读取「信息」APP 好友列表（ios-im 库 kv 表 im.friends，localStorage 兜底；同源可直读） */
   503→function loadChatFriends() {
   504→  const fromLS = () => {
   505→    try {
   506→      const raw = localStorage.getItem('im.friends');
   507→      const arr = raw ? JSON.parse(raw) : [];
   508→      return (Array.isArray(arr) ? arr : []).filter(f => f && f.name).map(f => ({ id: f.id, name: f.name }));
   509→    } catch (e) { return []; }
   510→  };
   511→  return new Promise((resolve) => {
   512→    let done = false;
   513→    const finish = (v) => { if (!done) { done = true; resolve(v); } };
   514→    try {
   515→      const req = indexedDB.open('ios-im');
   516→      req.onupgradeneeded = () => {
   517→        try { if (!req.result.objectStoreNames.contains('kv')) req.result.createObjectStore('kv'); } catch (e) { /* noop */ }
   518→      };
   519→      req.onsuccess = () => {
   520→        const db = req.result;
   521→        try {
   522→          const r = db.transaction('kv', 'readonly').objectStore('kv').get('im.friends');
   523→          r.onsuccess = () => {
   524→            const arr = Array.isArray(r.result) ? r.result : null;
   525→            const list = (arr || []).filter(f => f && f.name).map(f => ({ id: f.id, name: f.name }));
   526→            finish(list.length ? list : fromLS());
   527→          };
   528→          r.onerror = () => finish(fromLS());
   529→        } catch (e) { finish(fromLS()); }
   530→      };
   531→      req.onerror = () => finish(fromLS());
   532→      setTimeout(() => finish(fromLS()), 1500); /* 兜底超时 */
   533→    } catch (e) { finish(fromLS()); }
   534→  });
   535→}
