    95→function initialState(): { tab: Tab; view: View } {
    96→  const saved = loadUiState<{ tab?: Tab; view?: View }>({})
    97→  const tab: Tab = TABS.some((t) => t.key === saved.tab) ? (saved.tab as Tab) : 'messages'
    98→  const savedView = saved.view as View | undefined
    99→  let view: View = { name: 'tabs' }
   100→  if (savedView && typeof savedView.name === 'string') {
   101→    if (savedView.name === 'chat') {
   102→      view = loadFriends().some((f) => f.id === savedView.friendId) ? savedView : { name: 'tabs' }
   103→    } else if (savedView.name === 'stickers') {
   104→      view = savedView
   105→    } else if (savedView.name === 'billDetail') {
   106→      view = loadBills().some((b) => b.id === (savedView as { billId: string }).billId) ? savedView : { name: 'tabs' }
   107→    } else if (savedView.name === 'redPacket' || savedView.name === 'transfer') {
   108→      view = savedView.friendId && loadFriends().some((f) => f.id === savedView.friendId) ? savedView : { name: 'tabs' }
   109→    } else if (WALLET_VIEWS.includes(savedView.name)) {
   110→      view = savedView
   111→    } else if (
   112→      ['moments', 'myProfile', 'settings', 'apiSetting', 'visionApi', 'voiceApi', 'memory', 'chatSettings', 'chatSearch', 'chatBg', 'translateLang'].includes(
   113→        savedView.name
   114→      ) &&
   115→      loadFriends().some((f) => f.id === (savedView as { friendId: string }).friendId)
   116→    ) {
   117→      view = savedView
   118→    }
   119→  }
   120→  return { tab, view }
   121→}

   140→  /* 底部上滑手势（仅壳层内嵌时生效）：在屏幕底部 Home 区上滑 → 上报壳层打开多任务切换器，
   141→     与壳层里其他原生 App 的上滑唤起切换器体验保持一致（iframe 内手势无法冒泡到壳层页面）
   142→     判定策略：以轨迹最高点（minY）衡量上滑进度，只在其明显回落时才判定为列表滚动而取消，
   143→     避免起手抖动、慢速滑动被误杀
   144→     事件绑定（双流，tracking 标志天然去重）：
   145→     - 鼠标/触控笔：Pointer 事件（桌面可拖拽测试）
   146→     - 触摸：touch 事件 —— 真机上浏览器接管列表滚动时会派发 pointercancel 而不再派发
   147→       pointermove，但 touchmove 始终触发；触摸流不走 Pointer 事件，防止被 pointercancel 中止 */
   148→  useEffect(() => {
   149→    if (!embedded) return
   150→    const EDGE = 84 // 底部起手区高度
   151→    const MIN_UP = 24 // 有效上滑位移（相对轨迹最高点）
   152→    const BACK_GIVE = 30 // 冲高后回落超过该值 → 视为向下滑动，放弃
   153→    const MAX_DX = 110 // 允许的横向偏移
   154→    const MAX_MS = 1100 // 手势时间窗
   155→    let sx = 0
   156→    let sy = 0
   157→    let minY = 0
   158→    let st = 0
   159→    let tracking = false
   160→    let fired = false
   161→
   162→    const begin = (x: number, y: number, t: number) => {
   163→      if (fired || tracking) return
   164→      if (y < window.innerHeight - EDGE) return
   165→      tracking = true
   166→      sx = x
   167→      sy = y
   168→      minY = y
   169→      st = t
   170→    }
   171→    const move = (x: number, y: number, t: number) => {
   172→      if (!tracking || fired) return
   173→      if (y < minY) minY = y
   174→      if (y - minY > BACK_GIVE) {
   175→        tracking = false
   176→        return
   177→      }
   178→      const dx = Math.abs(x - sx)
   179→      if (sy - minY > MIN_UP && dx < MAX_DX && t - st < MAX_MS) {
   180→        fired = true
   181→        tracking = false
   182→        try {
   183→          window.parent.postMessage({ type: 'ios-open-switcher' }, window.location.origin)
   184→        } catch {
   185→          /* 跨域忽略 */
   186→        }
   187→        window.setTimeout(() => {
   188→          fired = false
   189→        }, 200)
   190→      }
   191→    }
   192→    const end = () => {
   193→      tracking = false
   194→    }
   195→
   196→    /* 鼠标/触控笔流：Pointer 事件 */
   197→    const onPd = (e: PointerEvent) => {
   198→      if (e.isPrimary && (e.pointerType === 'mouse' || e.pointerType === 'pen')) begin(e.clientX, e.clientY, e.timeStamp)
   199→    }
   200→    const onPm = (e: PointerEvent) => {
   201→      if (e.pointerType === 'mouse' || e.pointerType === 'pen') move(e.clientX, e.clientY, e.timeStamp)
   202→    }
   203→    /* 触摸指针的 pointerup/pointercancel 不在此收尾：浏览器接管滚动时会先发 pointercancel
   204→       （此时 touch 流仍在继续，touchmove 仍持续派发），触摸流由 touchend/touchcancel 收尾
   205→       —— 与壳层 onSwipe 的双流策略完全对齐（壳层根本不监听 pointer 流的触摸事件） */
   206→    const onPu = (e: PointerEvent) => {
   207→      if (e.pointerType === 'touch') return
   208→      end()
   209→    }
   210→
   211→    /* 触摸流：touch 事件（浏览器接管滚动后仍持续派发，pointercancel 不影响本流） */
   212→    const onTs = (e: TouchEvent) => {
   213→      if (e.touches.length === 1) begin(e.touches[0].clientX, e.touches[0].clientY, e.timeStamp)
   214→    }
   215→    const onTm = (e: TouchEvent) => {
   216→      if (e.touches.length === 1) move(e.touches[0].clientX, e.touches[0].clientY, e.timeStamp)
   217→    }
   218→    const onTe = () => end()
   219→
   220→    window.addEventListener('pointerdown', onPd)
   221→    window.addEventListener('pointermove', onPm)
   222→    window.addEventListener('pointerup', onPu)
   223→    window.addEventListener('pointercancel', onPu)
   224→    window.addEventListener('touchstart', onTs, { passive: true })
   225→    window.addEventListener('touchmove', onTm, { passive: true })
   226→    window.addEventListener('touchend', onTe, { passive: true })
   227→    window.addEventListener('touchcancel', onTe, { passive: true })
   228→    return () => {
   229→      window.removeEventListener('pointerdown', onPd)
   230→      window.removeEventListener('pointermove', onPm)
   231→      window.removeEventListener('pointerup', onPu)
   232→      window.removeEventListener('pointercancel', onPu)
   233→      window.removeEventListener('touchstart', onTs)
   234→      window.removeEventListener('touchmove', onTm)
   235→      window.removeEventListener('touchend', onTe)
   236→      window.removeEventListener('touchcancel', onTe)
   237→    }
   238→  }, [embedded])

   240→  /* 视图历史栈：壳层返回键逐级回退；显式 onBack 与栈自动对齐（目标===栈顶时弹栈） */

   374→  const submitRedPacket = (friendId: string, amount: number, blessing: string, payId: string, pwd: string | null): string | null => {
   375→    const friend = loadFriends().find((f) => f.id === friendId)
   376→    if (!friend) return '好友不存在'
   377→    const n = Math.round(amount * 100) / 100
   378→    if (!n || n <= 0 || n > 200) return '单个红包金额不可超过 200 元'
   379→    const w = loadWallet()
   380→    const a = checkAmount(w, payId, n)
   381→    if (a) return a
   382→    const v = verifyPwd(pwd)
   383→    if (v) return v
   384→    const label = payLabel(w, payId)
   385→    updateWallet((x) => deduct(x, payId, n))
   386→    addBill({ kind: '红包', title: `发给${friend.name}的红包`, amount: -n, status: '已发出，等待领取', friendName: friend.name, note: blessing })
   387→    const msg: Message = { id: uid(), friendId, from: 'me', text: `[红包:${blessing}|${n}]`, time: Date.now(), redpacket: { amount: n, blessing, status: '待领取' } }
   388→    appendMessage(msg)
   389→    refreshData()
   390→    scheduleAIReceive(friendId, msg.id, 'redpacket')
   391→    setView({
   392→      name: 'paySuccess',
   393→      data: {
   394→        title: '红包已发出',
   395→        amount: n,
   396→        rows: [
   397→          { label: '收款人', value: friend.name },
   398→          { label: '祝福语', value: blessing || '恭喜发财，大吉大利' },
   399→          { label: '支付方式', value: label },
   400→        ],
   401→        back: { name: 'chat', friendId },
   402→      },
   403→    })
   404→    return null
   405→  }

   407→  const submitTransfer = (friendId: string, amount: number, note: string, payId: string, pwd: string | null): string | null => {
   408→    const friend = loadFriends().find((f) => f.id === friendId)
   409→    if (!friend) return '好友不存在'
   410→    const n = Math.round(amount * 100) / 100

   473→  const submitSpend = (cardId: string, amount: number, note: string, payId: string, pwd: string | null): string | null => {

   507→    const label = payLabel(w, payId)
   508→    updateWallet((x) => ({
   509→      ...deduct(x, payId, n),
   510→      relativeCards: x.relativeCards.map((c) => (c.id === cardId ? { ...c, used: Math.round((c.used + n) * 100) / 100 } : c)),
   511→    }))

   514→      name: 'paySuccess',

   520→          { label: '支付方式', value: label },

   523→      },
   524→    })

   843→   渲染单个设置页组件，存储与信息APP完全同一份（im.api）；
   844→   页面自身返回键隐藏（.phone.embedded），由壳层全局返回键统一收起子页 —— 与信息APP内嵌行为一致 */
   845→function standalonePage(): 'api' | 'vision' | 'voice' | null {
   846→  if (typeof window === 'undefined') return null
   847→  const params = new URLSearchParams(window.location.search)
   848→  if (params.get('as') !== 'page') return null
   849→  const p = params.get('p')
   850→  return p === 'api' || p === 'vision' || p === 'voice' ? p : null
   851→}
   852→
   853→export default function App() {
   854→  const [asApp] = useState(isAsApp)
   855→  const [page] = useState(standalonePage)
   856→  useEffect(() => {
   857→    if (!asApp && !page) document.title = '主屏幕'
   858→  }, [asApp, page])
   859→  if (page) {
   860→    /* 子页返回桥接：同源调用壳层设置应用弹出子页（返回键被隐藏时为安全兑底） */
   861→    const back = () => {
   862→      try {
   863→        ;(window.parent as unknown as { __covePageBack?: () => void }).__covePageBack?.()
   864→      } catch {
   865→        /* 跨域忽略 */
   866→      }
   867→    }
   868→    return (
   869→      <div className="phone embedded">
   870→        {page === 'api' && <ApiSettingPage onBack={back} />}
   871→        {page === 'vision' && <VisionApiPage onBack={back} />}
   872→        {page === 'voice' && <VoiceApiPage onBack={back} />}
   873→      </div>
   874→    )
   875→  }
   876→  if (asApp) return <AppView />
   877→  return (
   878→    <div style={{ position: 'fixed', inset: 0, width: '100%', height: '100%', background: '#000' }}>
   879→      <iframe
   880→        title="主屏幕"
   881→        src="/ios/index.html"
   882→        style={{ width: '100%', height: '100%', border: 0, display: 'block' }}
   883→      />
   884→    </div>
   885→  )
   886→}
