    95→function initialState(): { tab: Tab; view: View } {
    96→  const saved = loadUiState<{ tab?: Tab; view?: View }>({})
    97→  const tab: Tab = TABS.some((t) => t.key === saved.tab) ? (saved.tab as Tab) : 'messages'
    98→  const savedView = saved.view as View | undefined
    99→  let view: View = { name: 'tabs' }
   100→  if (savedView && typeof savedView.name === 'string') {
   101→    if (savedView.name === 'chat') {
   102→      view = loadFriends().some((f) => f.id === savedView.friendId) ? savedView : { name: 'tabs' }
   103→    } else if (savedView.name === 'stickers') {
   104→      view = savedView
   105→    } else if (savedView.name === 'billDetail') {
   106→      view = loadBills().some((b) => b.id === (savedView as { billId: string }).billId) ? savedView : { name: 'tabs' }
   107→    } else if (savedView.name === 'redPacket' || savedView.name === 'transfer') {
   108→      view = savedView.friendId && loadFriends().some((f) => f.id === savedView.friendId) ? savedView : { name: 'tabs' }
   109→    } else if (WALLET_VIEWS.includes(savedView.name)) {
   110→      view = savedView
   111→    } else if (
   112→      ['moments', 'myProfile', 'settings', 'apiSetting', 'visionApi', 'voiceApi', 'memory', 'chatSettings', 'chatSearch', 'chatBg', 'translateLang'].includes(
   113→        savedView.name
   114→      ) &&
   115→      loadFriends().some((f) => f.id === (savedView as { friendId: string }).friendId)
   116→    ) {
   117→      view = savedView
   118→    }
   119→  }
   120→  return { tab, view }
   121→}

   131→  /* 壳层内嵌（iPhone 桌面打开）：页面内返回键隐藏，由壳层全局返回键统一接管 */
   132→  const [embedded] = useState(() => {
   133→    try {

   135→    } catch {
   136→      return false

   140→  /* 底部上滑手势（仅壳层内嵌时生效）：在屏幕底部 Home 区上滑 → 上报壳层打开多任务切换器，
   141→     与壳层里其他原生 App 的上滑唤起切换器体验保持一致（iframe 内手势无法冒泡到壳层页面）
   142→     判定策略：以轨迹最高点（minY）衡量上滑进度，只在其明显回落时才判定为列表滚动而取消，
   143→     避免起手抖动、慢速滑动被误杀 */
   144→  useEffect(() => {
   145→    if (!embedded) return
   146→    const EDGE = 84 // 底部起手区高度
   147→    const MIN_UP = 24 // 有效上滑位移（相对轨迹最高点）
   148→    const BACK_GIVE = 30 // 冲高后回落超过该值 → 视为向下滑动，放弃
   149→    const MAX_DX = 110 // 允许的横向偏移
   150→    const MAX_MS = 1100 // 手势时间窗
   151→    let sx = 0
   152→    let sy = 0
   153→    let minY = 0
   154→    let st = 0
   155→    let tracking = false
   156→    let fired = false
   157→
   158→    const begin = (x: number, y: number, t: number) => {
   159→      if (fired) return
   160→      if (y < window.innerHeight - EDGE) return
   161→      tracking = true
   162→      sx = x
   163→      sy = y
   164→      minY = y
   165→      st = t
   166→    }
   167→    const move = (x: number, y: number, t: number) => {
   168→      if (!tracking || fired) return
   169→      if (y < minY) minY = y
   170→      if (y - minY > BACK_GIVE) {
   171→        tracking = false
   172→        return
   173→      }
   174→      const dx = Math.abs(x - sx)
   175→      if (sy - minY > MIN_UP && dx < MAX_DX && t - st < MAX_MS) {
   176→        fired = true
   177→        tracking = false
   178→        try {
   179→          window.parent.postMessage({ type: 'ios-open-switcher' }, window.location.origin)
   180→        } catch {
   181→          /* 跨域忽略 */
   182→        }
   183→        window.setTimeout(() => {
   184→          fired = false
   185→        }, 200)
   186→      }
   187→    }
   188→    const end = () => {
   189→      tracking = false
   190→    }
   191→
   192→    const onPd = (e: PointerEvent) => {
   193→      if (e.isPrimary) begin(e.clientX, e.clientY, e.timeStamp)
   194→    }
   195→    const onPm = (e: PointerEvent) => {
   196→      if (e.isPrimary) move(e.clientX, e.clientY, e.timeStamp)
   197→    }
   198→    const onPu = () => end()
   199→
   200→    window.addEventListener('pointerdown', onPd)
   201→    window.addEventListener('pointermove', onPm)
   202→    window.addEventListener('pointerup', onPu)
   203→    window.addEventListener('pointercancel', onPu)
   204→    return () => {
   205→      window.removeEventListener('pointerdown', onPd)
   206→      window.removeEventListener('pointermove', onPm)
   207→      window.removeEventListener('pointerup', onPu)
   208→      window.removeEventListener('pointercancel', onPu)
   209→    }
   210→  }, [embedded])
   211→
   212→  /* 视图历史栈：壳层返回键逐级回退；显式 onBack 与栈自动对齐（目标===栈顶时弹栈） */
   213→  const viewStack = useRef<View[]>([])
   214→  const skipStack = useRef(false)
   215→  const lastView = useRef<View>(init.view)
   216→
   217→  const sameView = (a: View, b: View) => JSON.stringify(a) === JSON.stringify(b)
   218→
   219→  useEffect(() => {
   220→    const prev = lastView.current
   221→    lastView.current = view
   222→    if (skipStack.current) {
   223→      skipStack.current = false
   224→      return
   225→    }
   226→    if (sameView(prev, view)) return
   227→    const top = viewStack.current[viewStack.current.length - 1]
   228→    if (top && sameView(top, view)) viewStack.current.pop()
   229→    else if (prev.name !== 'paySuccess') viewStack.current.push(prev)
   230→  }, [view])
   231→
   232→  const navBack = (): boolean => {
   233→    if (view.name === 'tabs') return false
   234→    const prev = viewStack.current.pop()
   235→    skipStack.current = true
   236→    refreshData()
   237→    setView(prev ?? { name: 'tabs' })
   238→    return true
   239→  }

   346→  const submitRedPacket = (friendId: string, amount: number, blessing: string, payId: string, pwd: string | null): string | null => {
   347→    const friend = loadFriends().find((f) => f.id === friendId)
   348→    if (!friend) return '好友不存在'
   349→    const n = Math.round(amount * 100) / 100
   350→    if (!n || n <= 0 || n > 200) return '单个红包金额不可超过 200 元'
   351→    const w = loadWallet()
   352→    const a = checkAmount(w, payId, n)
   353→    if (a) return a
   354→    const v = verifyPwd(pwd)
   355→    if (v) return v
   356→    const label = payLabel(w, payId)
   357→    updateWallet((x) => deduct(x, payId, n))
   358→    addBill({ kind: '红包', title: `发给${friend.name}的红包`, amount: -n, status: '已发出，等待领取', friendName: friend.name, note: blessing })
   359→    const msg: Message = { id: uid(), friendId, from: 'me', text: `[红包:${blessing}|${n}]`, time: Date.now(), redpacket: { amount: n, blessing, status: '待领取' } }
   360→    appendMessage(msg)
   361→    refreshData()
   362→    scheduleAIReceive(friendId, msg.id, 'redpacket')
   363→    setView({
   364→      name: 'paySuccess',
   365→      data: {
   366→        title: '红包已发出',
   367→        amount: n,
   368→        rows: [
   369→          { label: '收款人', value: friend.name },

   375→    })

   377→  }

   445→  const submitSpend = (cardId: string, amount: number, note: string, payId: string, pwd: string | null): string | null => {
   446→    const w = loadWallet()
   447→    const card = w.relativeCards.find((c) => c.id === cardId)
   448→    if (!card) return '亲属卡不存在'
   449→    if (card.status === 'rejected') return '该亲属卡已退还，无法使用'
   450→    if (card.direction === 'received' && card.status !== 'claimed') return '请先领用该亲属卡'
   451→    const n = Math.round(amount * 100) / 100
   452→    if (!n || n <= 0) return '请输入正确的金额'
   453→    const rest = Math.round((card.monthlyLimit - card.used) * 100) / 100
   454→    if (n > rest) return '超过本卡当月剩余额度'
   455→    const v = verifyPwd(pwd)
   456→    if (v) return v
   457→    if (card.direction === 'received') {
   458→      updateWallet((x) => ({
   459→        ...x,
   460→        relativeCards: x.relativeCards.map((c) => (c.id === cardId ? { ...c, used: Math.round((c.used + n) * 100) / 100 } : c)),
   461→      }))
   462→      addBill({ kind: '亲属卡', title: `${card.friendName} 的亲属卡消费`, amount: 0, status: `由${card.friendName}代付`, friendName: card.friendName, note: note.trim() || '亲属卡消费' })
   463→      setView({
   464→        name: 'paySuccess',
   465→        data: {
   466→          title: '支付成功',
   467→          amount: n,
   468→          rows: [
   469→            { label: '消费项目', value: `${card.friendName} 的亲属卡` },
   470→            { label: '代付人', value: card.friendName },
   471→          ],
   472→          back: { name: 'relativeCard' },
   473→        },
   474→      })
   475→      return null
   476→    }
   477→    const a = checkAmount(w, payId, n)
   478→    if (a) return a
   479→    const label = payLabel(w, payId)
   480→    updateWallet((x) => ({
   481→      ...deduct(x, payId, n),
   482→      relativeCards: x.relativeCards.map((c) => (c.id === cardId ? { ...c, used: Math.round((c.used + n) * 100) / 100 } : c)),
   483→    }))
   484→    addBill({ kind: '亲属卡', title: `${card.friendName} 的亲属卡消费`, amount: -n, status: `已用${label}支付`, friendName: card.friendName, note: note.trim() || '亲属卡消费' })
   485→    setView({
   486→      name: 'paySuccess',
   487→      data: {
   488→        title: '支付成功',
   489→        amount: n,
   490→        rows: [
   491→          { label: '消费项目', value: `${card.friendName} 的亲属卡` },
   492→          { label: '支付方式', value: label },
   493→        ],
   494→        back: { name: 'relativeCard' },
   495→      },
   496→    })
   497→    return null
   498→  }
